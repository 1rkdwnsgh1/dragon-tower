; 기본 UI 스킨 — 창틀·버튼·토글·막대 그림의 여백과 맞춤 방식
;
; [조각이름] 아래에 아래 값을 적는다. 적지 않으면 기본값이 쓰인다.
;
;   fit    = nine | height | contain      (기본 nine)
;   left/top/right/bottom = 늘리지 않을 테두리 두께 (원본 그림 기준 픽셀)
;
; fit 이 하는 일:
;   nine    — 9조각. 네 모서리를 지키고 가장자리·가운데만 늘린다. 큰 창틀용.
;             테두리가 목적지의 35%를 넘으면 그만큼 줄이고, 그래도 많이 줄여야 하면
;             그림을 쓰지 않고 도형으로 그린다. 큰 장식 프레임을 작은 창에 넣으면
;             테두리가 가운데를 다 먹어 장식만 남기 때문이다.
;   height  — 높이를 목적지에 맞춰 통째로 줄이고 가로만 늘린다. 세로로 눌리지 않는다.
;             버튼·막대처럼 낮고 가로로 긴 것에 쓴다. left/right 만 의미가 있다.
;   contain — 비율을 지킨 채 목적지 안에 넣는다. 토글·체크처럼 모양이 정해진 것.
;
; 여백을 바꾸기 전에 원본을 확대해서 장식이 어디까지 이어지는지 보고 정한다.
; 숫자만 키워도 작은 창 문제는 해결되지 않는다 — 그때는 fit 이나 그림 자체를 바꿔야 한다.

; ---------------------------------------------------------------------------
; 창틀 (1109x1382). 상단 왕관·날개와 하단 보석이 크다.
; 큰 창(선택 화면·모달)에서만 쓰이고, 상태창·메뉴처럼 작은 창에서는 위 규칙에 따라
; 자동으로 도형 그리기로 넘어간다. 작은 창용 얇은 프레임이 생기면 panel_small 로 넣는다.
; ---------------------------------------------------------------------------
[panel]
fit=nine
left=120
top=150
right=120
bottom=130

; ---------------------------------------------------------------------------
; 버튼 (402x106 / hover 426x137 / off 384x106)
; 세로로 눌리면 안 되므로 height. 양끝 장식 두께만 적는다.
; hover 는 발광 때문에 캔버스가 더 크다. 같은 자리에 그려도 높이에 맞춰 줄어든다.
; ---------------------------------------------------------------------------
[button]
fit=height
left=70
right=70

[button_hover]
fit=height
left=80
right=80

[button_off]
fit=height
left=68
right=68

; 좁은 자리의 작은 버튼 (241x81 / 246x97 / 241x82)
[pill]

right=50

[pill_hover]
fit=height

right=54

[pill_off]

right=50
